cmake_minimum_required(VERSION 3.25)
project(campervan_rtabmap_extension LANGUAGES CXX)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(RTABMap CONFIG REQUIRED)
find_package(OpenCV CONFIG REQUIRED)
find_package(PCL CONFIG REQUIRED COMPONENTS common io octree kdtree search surface filters registration sample_consensus segmentation)
find_package(g2o CONFIG REQUIRED)
find_package(CXSparse CONFIG REQUIRED)
Python_add_library(_native MODULE WITH_SOABI bridge.cpp "${TOOL_DIR}/Export.cpp" "${TOOL_DIR}/Reprocess.cpp")
set_source_files_properties("${TOOL_DIR}/Export.cpp" PROPERTIES COMPILE_DEFINITIONS "main=campervan_export_main")
set_source_files_properties("${TOOL_DIR}/Reprocess.cpp" PROPERTIES COMPILE_DEFINITIONS "main=campervan_reprocess_main")
target_include_directories(_native PRIVATE ${RTABMap_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
target_link_libraries(_native PRIVATE ${RTABMap_LIBRARIES} ${OpenCV_LIBS} ${PCL_LIBRARIES}
  g2o::core g2o::solver_csparse SuiteSparse::CXSparse_static)
target_compile_definitions(_native PRIVATE CAMPERVAN_BUILD_ID="${BUILD_ID}")
target_link_options(_native PRIVATE "LINKER:-exported_symbol,_PyInit__native")
install(TARGETS _native LIBRARY DESTINATION campervan_rtabmap)
